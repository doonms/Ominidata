// =====================================================
// MENU LATERAL — Ominidata (compatível com SPA)
// - Reanexa eventos após navegação (spa:page-loaded)
// - Resolve caminhos (root vs /pages/)
// - Mantém compatibilidade com criarMenuLateral/carregarMenuLateral
// =====================================================

import { getCurrentUser, isAdmin } from "../auth/accessControl.js";

function getBasePrefix() {
  // Se a página atual está em /pages/, usamos "../" como base; caso contrário, "./"
  const path = window.location.pathname || '';
  return /\/pages\/.+\.html$/i.test(path) ? '../' : './';
}

function buildLink(href) {
  // Monta link relativo ao local atual, respeitando /pages/
  const base = getBasePrefix();
  return base + href.replace(/^\.?\//, '');
}

function buildLogoSrc() {
  // Garante que o caminho do logo funciona tanto no index quanto em /pages/
  return buildLink('imagens/logos/logo.png');
}

export function criarMenuLateral() {
  const sidebar = document.getElementById('sidebar');
  if (!sidebar) return;

  const wasOpen = sidebar.classList.contains('open'); // preserva estado aberto/fechado
  const homeHref       = buildLink('index.html');
  const simuladorHref  = buildLink('pages/simulador.html');
  const fatoresHref    = buildLink('pages/base-fatores.html');
  const inssHref       = buildLink('pages/base-inss.html');
  const creditoHref    = buildLink('pages/base-credito-pessoal.html');
  const bancosHref     = buildLink('pages/bancos.html');
  const pesquisaHref   = buildLink('pages/pesquisa.html');
  const contatosHref   = buildLink('pages/contatos.html');

  sidebar.innerHTML = `
    <div class="menu-logo">
      <div class="menu-bg" aria-hidden="true"></div>
      <img src="${buildLogoSrc()}" class="menu-logo-img" alt="Logo" />
    </div>

    <a class="publico" href="${homeHref}" data-nav="home">🏠 Início</a>
    <a class="publico" href="${simuladorHref}" data-nav="simulador">💳 Simulador</a>
    <a class="admin-only" href="${fatoresHref}" data-nav="base_fatores">🧮 Base de Fatores</a>
    <a class="admin-only" href="${inssHref}" data-nav="base_inss">📁 Base INSS</a>
    <a class="admin-only" href="${creditoHref}" data-nav="base_credito">📁 Crédito Pessoal</a>
    <a class="publico" href="${bancosHref}" data-nav="bancos">🏦 Bancos</a>
    <a class="publico" href="${pesquisaHref}" data-nav="pesquisa">🔍 Pesquisa</a>
    <a class="publico" href="${contatosHref}" data-nav="contatos">📞 Contatos</a>
  `;

  // Rebind do botão hamburguer ☰ — usa .onclick para evitar múltiplos handlers
  const menuBtn = document.getElementById('menuBtn');
  if (menuBtn) {
    menuBtn.onclick = () => sidebar.classList.toggle('open');
  }

  // Fecha o menu quando clicar em qualquer link do próprio menu (delegação)
  // Garante um único handler reutilizável mesmo após innerHTML
  if (!sidebar.dataset.bound) {
    sidebar.addEventListener('click', (e) => {
      const a = e.target.closest('a');
      if (!a) return;
      // Fecha só para navegação interna
      const sameOrigin = (() => {
        try { return new URL(a.href, window.location.origin).origin === window.location.origin; }
        catch { return true; }
      })();
      if (sameOrigin) sidebar.classList.remove('open');
    });
    sidebar.dataset.bound = '1';
  }

  // Restaura estado aberto se estava aberto antes do re-render
  if (wasOpen) sidebar.classList.add('open');

  getCurrentUser()
    .then(user => {
      if (!isAdmin(user)) {
        sidebar.querySelectorAll('.admin-only').forEach(el => el.remove());
      }
    })
    .catch(err => {
      console.error("Falha ao obter usuário para controle de acesso:", err);
      sidebar.querySelectorAll('.admin-only').forEach(el => el.remove());
    });
}

// Alias para compatibilidade retroativa
export const carregarMenuLateral = criarMenuLateral;

// Reexecuta o setup após cada navegação do SPA
document.addEventListener('spa:page-loaded', () => {
  criarMenuLateral();
});

// Executa ao carregar o DOM (para acessos diretos aos HTMLs)
if (document.readyState !== 'loading') {
  criarMenuLateral();
} else {
  document.addEventListener('DOMContentLoaded', criarMenuLateral);
}
